backup_pipeline.bat 
:: Automated Backup & Network Replication Script
@echo off
set TIMESTAMP=%date:~-4%%date:~4,2%%date:~7,2%
set BACKUP_DIR=D:\Database_Backups
set NAS_DEST=\\192.168.1.100\Server_Archive

echo [INFO] Starting SQL Database Backup...
sqlcmd -S localhost -Q "BACKUP DATABASE [ProductionDB] TO DISK='%BACKUP_DIR%\ProdDB_%TIMESTAMP%.bak'"

echo [INFO] Replicating backup to NAS via Robocopy...
robocopy %BACKUP_DIR% %NAS_DEST% *.bak /MIR /Z /R:3 /W:5 /LOG:backup_log.txt

echo [SUCCESS] Backup & Mirroring Completed.
